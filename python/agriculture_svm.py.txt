import os
import cv2
import numpy as np
from sklearn.svm import SVC

# =========================================================
# CONFIGURATION
# =========================================================

DATASET_PATH = r"C:\Xilinx\Vivado\2024.2\UG_PROJECT\data"

TEST_IMAGE_PATH = r"C:\Xilinx\Vivado\2024.2\UG_PROJECT\data\paddy\pr14.jpg"

OUTPUT_PATH = r"C:\Xilinx\Vivado\2024.2"

IMG_H = 64
IMG_W = 64

IMG_PIXELS = IMG_H * IMG_W
FEATURES = IMG_PIXELS * 3

FIXED_SCALE = 10000

CLASSES = [
    "mirchi",
    "cotton",
    "paddy",
    "sunflower"
]

# =========================================================
# HEX FUNCTION
# =========================================================

def to_hex_signed(val, bits=32):

    if val < 0:
        val = (1 << bits) + val

    return format(
        val & ((1 << bits) - 1),
        f'0{bits//4}X'
    )

# =========================================================
# LOAD DATASET
# =========================================================

def load_dataset(path):

    X = []
    y = []

    for label, folder in enumerate(CLASSES):

        folder_path = os.path.join(path, folder)

        if not os.path.exists(folder_path):
            continue

        for file in os.listdir(folder_path):

            img_path = os.path.join(
                folder_path,
                file
            )

            img = cv2.imread(img_path)

            if img is None:
                continue

            img = cv2.resize(
                img,
                (IMG_W, IMG_H)
            )

            img = cv2.cvtColor(
                img,
                cv2.COLOR_BGR2RGB
            )

            feature = (
                img.flatten()
                .astype(np.float32)
            ) / 255.0

            X.append(feature)
            y.append(label)

    return np.array(X), np.array(y)

# =========================================================
# LOAD DATASET
# =========================================================

print("Loading Dataset...")

X, y = load_dataset(DATASET_PATH)

print(
    f"Dataset Loaded - "
    f"Samples: {len(X)}, "
    f"Features: {X.shape[1]}"
)

# =========================================================
# TRAIN SVM
# =========================================================

print("\nTraining SVM (OvO)...")

model = SVC(
    kernel='linear',
    decision_function_shape='ovo'
)

model.fit(X, y)

print("Training Completed")

# =========================================================
# TEST IMAGE
# =========================================================

test_img = cv2.imread(TEST_IMAGE_PATH)

test_img = cv2.resize(
    test_img,
    (IMG_W, IMG_H)
)

test_rgb = cv2.cvtColor(
    test_img,
    cv2.COLOR_BGR2RGB
)

test_feature = (
    test_rgb.flatten()
    .astype(np.float32)
) / 255.0

# =========================================================
# SOFTWARE PREDICTION
# =========================================================

prediction = model.predict(
    test_feature.reshape(1, -1)
)[0]

crop_name = CLASSES[prediction]

decision = model.decision_function(
    test_feature.reshape(1, -1)
)

print("\n===================================")
print("SOFTWARE PREDICTION")
print("===================================")

print(f"Predicted Crop : {crop_name}")

print(
    f"Decision Function :\n"
    f"{decision[0]}"
)

# =========================================================
# RGB BASED YIELD ESTIMATION
# SAME AS VERILOG
# =========================================================

r = test_rgb[:, :, 0]
g = test_rgb[:, :, 1]
b = test_rgb[:, :, 2]

# ---------------------------------------------------------
# MIRCHI
# ---------------------------------------------------------

if crop_name == "mirchi":

    mask = (
        (
            (r > 100) &
            (r > g) &
            (r > b)
        )
        |
        (
            (g > 80) &
            (g > r) &
            (g > b)
        )
    )

# ---------------------------------------------------------
# COTTON
# ---------------------------------------------------------

elif crop_name == "cotton":

    mask = (
        (r > 140) &
        (g > 140) &
        (b > 140)
    )

# ---------------------------------------------------------
# PADDY
# ---------------------------------------------------------

elif crop_name == "paddy":

    mask = (
        (g > 80) &
        (g > r) &
        (g > b)
    )

# ---------------------------------------------------------
# SUNFLOWER
# ---------------------------------------------------------

elif crop_name == "sunflower":

    mask = (
        (r > 120) &
        (g > 120) &
        (b < 120)
    )

# ---------------------------------------------------------
# DEFAULT
# ---------------------------------------------------------

else:

    mask = (
        (g > r) &
        (g > b)
    )

# =========================================================
# YIELD %
# =========================================================

yield_percentage = (
    np.count_nonzero(mask)
    / IMG_PIXELS
) * 100

print(
    f"Estimated Yield : "
    f"{yield_percentage:.2f}%"
)

# =========================================================
# EXPORT TEST IMAGE
# =========================================================

print("\nExporting Files...")

with open(
    os.path.join(
        OUTPUT_PATH,
        "test_image.txt"
    ),
    "w"
) as f:

    for pixel in test_rgb.reshape(-1, 3):

        r1 = int(pixel[0])
        g1 = int(pixel[1])
        b1 = int(pixel[2])

        rgb24 = (
            (r1 << 16) |
            (g1 << 8) |
            b1
        )

        f.write(
            format(rgb24, '06X') + "\n"
        )

# =========================================================
# EXPORT FEATURES
# =========================================================

test_fixed = np.clip(

    (
        test_feature *
        FIXED_SCALE
    ).astype(np.int64),

    -2**31,
    2**31 - 1

).astype(np.int32)

with open(
    os.path.join(
        OUTPUT_PATH,
        "feature_mem.txt"
    ),
    "w"
) as f:

    for val in test_fixed:

        f.write(
            to_hex_signed(
                int(val),
                32
            ) + "\n"
        )

# =========================================================
# EXPORT WEIGHTS
# =========================================================

weights = np.clip(

    (
        model.coef_ *
        FIXED_SCALE
    ).astype(np.int64),

    -2**31,
    2**31 - 1

).astype(np.int32)

with open(
    os.path.join(
        OUTPUT_PATH,
        "svm_weights.txt"
    ),
    "w"
) as f:

    for clf in weights:

        for w in clf:

            f.write(
                to_hex_signed(
                    int(w),
                    32
                ) + "\n"
            )

# =========================================================
# EXPORT BIAS
# =========================================================

bias = np.clip(

    (
        model.intercept_ *
        FIXED_SCALE
    ).astype(np.int64),

    -2**31,
    2**31 - 1

).astype(np.int32)

with open(
    os.path.join(
        OUTPUT_PATH,
        "svm_bias.txt"
    ),
    "w"
) as f:

    for b in bias:

        f.write(
            to_hex_signed(
                int(b),
                32
            ) + "\n"
        )

print("\n===================================")
print("FILES GENERATED SUCCESSFULLY")
print("===================================")
